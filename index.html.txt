        /* ================= TAB NAVIGATION ================= */
        function switchTab(tabId) {
            activeTab = tabId;
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById(`tab-${tabId}`).classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('text-indigo-400', 'bg-indigo-500/10');
                btn.classList.add('text-slate-300');
            });
            const activeNavBtn = document.getElementById(`nav-${tabId}`);
            if (activeNavBtn) {
                activeNavBtn.classList.remove('text-slate-300');
                activeNavBtn.classList.add('text-indigo-400', 'bg-indigo-500/10');
            }

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function toggleMobileMenu() {
            const menu = document.getElementById('mobileMenu');
            menu.classList.toggle('hidden');
        }

        /* ================= RENDER FUNCTIONS ================= */
        function renderGradePills() {
            const container = document.getElementById('gradePillsContainer');
            let html = `<button onclick="filterByGrade(null)" class="px-4 py-2 rounded-xl text-xs font-semibold whitespace-nowrap bg-indigo-600 text-white shadow-sm">සියලුම ශ්‍රේණි</button>`;
            for (let i = 1; i <= 13; i++) {
                html += `
                    <button onclick="filterByGrade(${i})" class="grade-pill-${i} px-4 py-2 rounded-xl text-xs font-semibold whitespace-nowrap bg-slate-900 border border-slate-800 text-slate-300 hover:text-white hover:border-indigo-500 transition-all">
                        ${i} ශ්‍රේණිය
                    </button>
                `;
            }
            container.innerHTML = html;
        }

        function renderHomeContent() {
            // Render Featured Papers (First 6)
            const homePapersGrid = document.getElementById('homePapersGrid');
            const featuredPapers = samplePapers.slice(0, 6);
            homePapersGrid.innerHTML = featuredPapers.map(p => createPaperCardHTML(p)).join('');

            // Render Featured Textbooks
            const homeTextbooksGrid = document.getElementById('homeTextbooksGrid');
            const featuredBooks = sampleTextbooks.slice(0, 6);
            homeTextbooksGrid.innerHTML = featuredBooks.map(b => createTextbookCardHTML(b)).join('');
        }

        function renderPapersGrid() {
            const grid = document.getElementById('allPapersGrid');
            let filtered = samplePapers;

            if (selectedPaperCategory !== 'ALL') {
                filtered = filtered.filter(p => p.category === selectedPaperCategory);
            }
            if (selectedGradeFilter) {
                filtered = filtered.filter(p => p.grade === selectedGradeFilter);
            }

            if (filtered.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-12 text-slate-500">ප්‍රශ්න පත්‍ර හමුවූයේ නැත.</div>`;
                return;
            }

            grid.innerHTML = filtered.map(p => createPaperCardHTML(p)).join('');
        }

        function renderTextbooksGrid() {
            const grid = document.getElementById('allTextbooksGrid');
            let filtered = sampleTextbooks.filter(b => b.medium === currentMedium);

            if (selectedGradeFilter) {
                filtered = filtered.filter(b => b.grade === selectedGradeFilter);
            }

            if (filtered.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-12 text-slate-500">පෙළපොත් හමුවූයේ නැත.</div>`;
                return;
            }

            grid.innerHTML = filtered.map(b => createTextbookCardHTML(b)).join('');
        }

        /* ================= FILTER LOGIC ================= */
        function filterGradeCategory(cat) {
            if (cat === 'primary') filterByGrade(5);
            else if (cat === 'junior') filterByGrade(8);
            else if (cat === 'ol') filterByGrade(11);
            else if (cat === 'al') filterByGrade(13);
            switchTab('papers');
        }

        function filterByGrade(grade) {
            selectedGradeFilter = grade;
            showToast(grade ? `${grade} ශ්‍රේණිය තෝරාගන්නා ලදී` : 'සියලුම ශ්‍රේණි පෙන්වයි');
            renderPapersGrid();
            renderTextbooksGrid();
        }

        function filterPapersCategory(cat) {
            selectedPaperCategory = cat;
            document.querySelectorAll('.paper-cat-pill').forEach(btn => {
                btn.classList.remove('bg-indigo-600', 'text-white');
                btn.classList.add('bg-slate-900', 'border', 'border-slate-800', 'text-slate-300');
            });
            event.target.classList.remove('bg-slate-900', 'border', 'border-slate-800', 'text-slate-300');
            event.target.classList.add('bg-indigo-600', 'text-white');
            renderPapersGrid();
        }

        function filterMedium(med) {
            currentMedium = med;
            document.querySelectorAll('.medium-btn').forEach(btn => {
                btn.classList.remove('bg-indigo-600', 'text-white');
                btn.classList.add('text-slate-400');
            });
            const selectedBtn = document.getElementById(`btn-med-${med}`);
            selectedBtn.classList.remove('text-slate-400');
            selectedBtn.classList.add('bg-indigo-600', 'text-white');
            renderTextbooksGrid();
        }

        function filterBySubject(sub) {
            if (sub === 'සියල්ල') {
                renderPapersGrid();
                return;
            }
            const filtered = samplePapers.filter(p => p.subject === sub);
            const grid = document.getElementById('allPapersGrid');
            grid.innerHTML = filtered.map(p => createPaperCardHTML(p)).join('');
        }

        function handleGlobalSearch() {
            const query = event.target.value.toLowerCase().trim();
            if(!query) {
                renderPapersGrid();
                return;
            }
            const filtered = samplePapers.filter(p => 
                p.title.toLowerCase().includes(query) || 
                p.subject.toLowerCase().includes(query)
            );
            const grid = document.getElementById('allPapersGrid');
            grid.innerHTML = filtered.length ? filtered.map(p => createPaperCardHTML(p)).join('') : `<div class="col-span-full text-center py-12 text-slate-500">ප්‍රතිඵල හමුවූයේ නැත.</div>`;
        }

        /* ================= MCQ QUIZ ENGINE ================= */
        function startQuiz(quizKey) {
            activeQuiz = quizDataStore[quizKey];
            activeQuizState.currentQIdx = 0;
            activeQuizState.userAnswers = [];
            activeQuizState.timeLeft = activeQuiz.timer;

            document.getElementById('quizSelectionScreen').classList.add('hidden');
            document.getElementById('quizResultBox').classList.add('hidden');
            document.getElementById('activeQuizBox').classList.remove('hidden');

            document.getElementById('quizTitle').innerText = activeQuiz.title;

            startQuizTimer();
            renderCurrentQuestion();
        }

        function renderCurrentQuestion() {
            const qData = activeQuiz.questions[activeQuizState.currentQIdx];
            const totalQ = activeQuiz.questions.length;

            document.getElementById('quizQuestionNum').innerText = `ප්‍රශ්න අංක ${activeQuizState.currentQIdx + 1} / ${totalQ}`;
            document.getElementById('quizProgressBar').style.width = `${((activeQuizState.currentQIdx + 1) / totalQ) * 100}%`;
            document.getElementById('quizQuestionText').innerText = qData.q;

            const optionsContainer = document.getElementById('quizOptionsContainer');
            optionsContainer.innerHTML = qData.options.map((opt, idx) => `
                <button onclick="selectQuizOption(${idx})" class="quiz-opt-btn w-full p-4 rounded-xl border border-slate-800 bg-slate-950 hover:bg-slate-800/80 text-left text-sm text-slate-200 transition-all flex items-center justify-between">
                    <span>${idx + 1}. ${opt}</span>
                    <i class="fa-regular fa-circle text-slate-600 opt-check"></i>
                </button>
            `).join('');

            document.getElementById('nextQuestionBtn').disabled = true;
        }

        function selectQuizOption(optIdx) {
            activeQuizState.userAnswers[activeQuizState.currentQIdx] = optIdx;

            const buttons = document.querySelectorAll('.quiz-opt-btn');
            buttons.forEach((btn, idx) => {
                if (idx === optIdx) {
                    btn.classList.add('border-indigo-500', 'bg-indigo-500/10', 'text-white');
                    btn.querySelector('.opt-check').className = 'fa-solid fa-circle-check text-indigo-400';
                } else {
                    btn.classList.remove('border-indigo-500', 'bg-indigo-500/10', 'text-white');
                    btn.querySelector('.opt-check').className = 'fa-regular fa-circle text-slate-600';
                }
            });

            document.getElementById('nextQuestionBtn').disabled = false;
        }

        function nextQuestion() {
            if (activeQuizState.currentQIdx < activeQuiz.questions.length - 1) {
                activeQuizState.currentQIdx++;
                renderCurrentQuestion();
            } else {
                finishQuiz();
            }
        }

        function startQuizTimer() {
            clearInterval(activeQuizState.timerInterval);
            activeQuizState.timerInterval = setInterval(() => {
                activeQuizState.timeLeft--;
                const mins = Math.floor(activeQuizState.timeLeft / 60);
                const secs = activeQuizState.timeLeft % 60;
                document.getElementById('quizTimer').innerText = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;

                if (activeQuizState.timeLeft <= 0) {
                    clearInterval(activeQuizState.timerInterval);
                    finishQuiz();
                }
            }, 1000);
        }

        function finishQuiz() {
            clearInterval(activeQuizState.timerInterval);
            document.getElementById('activeQuizBox').classList.add('hidden');
            document.getElementById('quizResultBox').classList.remove('hidden');

            let score = 0;
            const total = activeQuiz.questions.length;

            const reviewContainer = document.getElementById('quizReviewContainer');
            reviewContainer.innerHTML = '';

            activeQuiz.questions.forEach((q, idx) => {
                const uAns = activeQuizState.userAnswers[idx];
                const isCorrect = uAns === q.answer;
                if (isCorrect) score++;

                reviewContainer.innerHTML += `
                    <div class="p-4 rounded-xl border ${isCorrect ? 'border-emerald-500/30 bg-emerald-950/20' : 'border-rose-500/30 bg-rose-950/20'} space-y-2">
                        <div class="flex items-center justify-between text-xs font-bold">
                            <span class="${isCorrect ? 'text-emerald-400' : 'text-rose-400'}">${idx + 1}. ${isCorrect ? 'නිවැරදියි' : 'වැරදියි'}</span>
                        </div>
                        <p class="text-sm font-semibold text-white">${q.q}</p>
                        <p class="text-xs text-slate-300">ඔබේ පිළිතුර: <span class="font-bold">${uAns !== undefined ? q.options[uAns] : 'පිළිතුරු දී නැත'}</span></p>
                        ${!isCorrect ? `<p class="text-xs text-emerald-400">නිවැරදි පිළිතුර: <span class="font-bold">${q.options[q.answer]}</span></p>` : ''}
                        <p class="text-xs text-slate-400 italic pt-1 border-t border-slate-800">පැහැදිලි කිරීම: ${q.explanation}</p>
                    </div>
                `;
            });

            document.getElementById('finalScoreText').innerText = `${score} / ${total}`;
            document.getElementById('scorePercentage').innerText = `ලකුණු ප්‍රතිශතය: ${Math.round((score / total) * 100)}%`;
        }

        function exitQuiz() {
            clearInterval(activeQuizState.timerInterval);
            resetQuizScreen();
        }

        }
