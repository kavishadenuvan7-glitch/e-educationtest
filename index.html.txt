        /* ==========================================================================
           2. INDEXEDDB SETUP FOR OFFLINE FILE STORAGE
           ========================================================================== */
        let db;
        function initIndexedDB() {
            return new Promise((resolve, reject) => {
                const request = indexedDB.open('SmartPassPapersDB', 1);
                request.onupgradeneeded = (e) => {
                    const db = e.target.result;
                    if (!db.objectStoreNames.contains('resources')) {
                        db.createObjectStore('resources', { keyPath: 'id' });
                    }
                };
                request.onsuccess = (e) => {
                    db = e.target.result;
                    resolve(db);
                };
                request.onerror = (e) => reject(e);
            });
        }

        async function getSavedResources() {
            if (!db) await initIndexedDB();
            return new Promise((resolve) => {
                const tx = db.transaction('resources', 'readonly');
                const store = tx.objectStore('resources');
                const request = store.getAll();
                request.onsuccess = () => resolve(request.result || []);
            });
        }

        async function saveResourceToDB(resource, blobFile) {
            if (!db) await initIndexedDB();
            return new Promise((resolve, reject) => {
                const tx = db.transaction('resources', 'readwrite');
                const store = tx.objectStore('resources');
                
                if (blobFile) {
                    resource.blob = blobFile;
                }
                
                const request = store.put(resource);
                request.onsuccess = () => resolve(true);
                request.onerror = () => reject(false);
            });
        }

        /* --- PAPERS FILTERING --- */
        function applyPaperFilters() {
            const grade = document.getElementById('paperGradeFilter').value;
            const subject = document.getElementById('paperSubjectFilter').value;
            const type = document.getElementById('paperTypeFilter').value;
            const medium = document.getElementById('paperMediumFilter').value;
            const sort = document.getElementById('paperSortFilter').value;
            const search = document.getElementById('globalSearchInput').value.toLowerCase();

            let filtered = allPapersList.filter(item => {
                if (grade !== 'ALL' && item.grade !== grade) return false;
                if (subject !== 'ALL' && item.subject !== subject) return false;
                if (type !== 'ALL' && item.type !== type) return false;
                if (medium !== 'ALL' && item.medium !== medium) return false;
                if (search && !item.title.toLowerCase().includes(search) && !item.subject.toLowerCase().includes(search)) return false;
                return true;
            });

            // Sort
            if (sort === 'title') filtered.sort((a,b) => a.title.localeCompare(b.title));
            else if (sort === 'oldest') filtered.sort((a,b) => a.id.localeCompare(b.id));
            else if (sort === 'downloads') filtered.sort((a,b) => b.downloads - a.downloads);

            const container = document.getElementById('papersContainer');
            const emptyState = document.getElementById('papersEmptyState');

            if (filtered.length === 0) {
                container.innerHTML = '';
                emptyState.classList.remove('hidden');
            } else {
                emptyState.classList.add('hidden');
                container.innerHTML = filtered.map(p => createPaperCardHTML(p)).join('');
            }
        }

        async function previewPDF(id) {
            const item = [...allPapersList, ...allBooksList].find(r => r.id === id);
            if (!item) return;

            document.getElementById('modalPdfTitle').textContent = item.title;
            document.getElementById('modalPdfSubtitle').textContent = `${item.subject} • Grade ${item.grade}`;
            
            const iframe = document.getElementById('pdfIframe');
            const fallback = document.getElementById('pdfFallback');
            const dlBtn = document.getElementById('modalDownloadBtn');
            const fallbackDlBtn = document.getElementById('modalFallbackDownload');

            dlBtn.onclick = () => downloadPDF(id);
            fallbackDlBtn.onclick = () => downloadPDF(id);

            document.getElementById('pdfModal').classList.remove('hidden');

            if (item.blob) {
                const blobUrl = URL.createObjectURL(item.blob);
                iframe.src = blobUrl;
                iframe.classList.remove('hidden');
                fallback.classList.add('hidden');
            } else if (item.pdfUrl) {
                iframe.src = item.pdfUrl;
                iframe.classList.remove('hidden');
                fallback.classList.add('hidden');
            } else {
                iframe.classList.add('hidden');
                fallback.classList.remove('hidden');
            }
        }

        function closePdfModal() {
            document.getElementById('pdfModal').classList.add('hidden');
            document.getElementById('pdfIframe').src = '';
        }

        async function downloadPDF(id) {
            const item = [...allPapersList, ...allBooksList].find(r => r.id === id);
            if (!item) return;

            item.downloads = (item.downloads || 0) + 1;
            
            // Record in download history
            downloadHistory.unshift({
                id: item.id,
                title: item.title,
                time: new Date().toLocaleString()
            });
            downloadHistory = downloadHistory.slice(0, 10);
            localStorage.setItem('spp_downloads', JSON.stringify(downloadHistory));

            showToast(`Downloading: ${item.title}`);
            renderDashboard();

            // Trigger Download File Blob or URL
            if (item.blob) {
                const link = document.createElement('a');
                link.href = URL.createObjectURL(item.blob);
                link.download = `${item.title.replace(/\s+/g, '_')}.pdf`;
                link.click();
            } else {
                window.open(item.pdfUrl || 'https://www.w3.org/WAI/ER/tests/xhtml/testfiles/resources/pdf/dummy.pdf', '_blank');
            }
        }

        async function handleResourceSubmit(e) {
            e.preventDefault();
            
            const title = document.getElementById('resTitle').value;
            const category = document.getElementById('resCategory').value;
            const subtype = document.getElementById('resSubtype').value;
            const grade = document.getElementById('resGrade').value;
            const subject = document.getElementById('resSubject').value;
            const medium = document.getElementById('resMedium').value;
            const school = document.getElementById('resSchoolYear').value;
            const description = document.getElementById('resDescription').value;
            
            const fileInput = document.getElementById('resPdfFile');
            const urlInput = document.getElementById('resPdfUrl').value;

            if (!fileInput.files[0] && !urlInput) {
                showToast('Please provide either a PDF File upload or external URL', 'error');
                return;
            }

            const newId = 'user-' + Date.now();
            const newRes = {
                id: newId,
                title,
                category,
                type: subtype,
                grade,
                subject,
                medium,
                school,
                publisher: school,
                description,
                pdfUrl: urlInput || null,
                downloads: 0,
                addedDate: new Date().toISOString().split('T')[0]
            };

            const blob = fileInput.files[0] || null;

            await saveResourceToDB(newRes, blob);
            showToast('Resource uploaded and saved successfully!', 'success');
            
            resetAddForm();
            await loadAllData();
            switchTab(category === 'Paper' ? 'papers' : 'textbooks');
        }

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Keyboard Shortcut ('/') for Quick Search
        document.addEventListener('keydown', (e) => {
            if (e.key === '/' && document.activeElement.tagName !== 'INPUT' && document.activeElement.tagName !== 'TEXTAREA') {
                e.preventDefault();
                const search = document.getElementById('globalSearchInput');
                if (search) search.focus();
            }
        });

        // Global Search Input Event Listener
        document.getElementById('globalSearchInput')?.addEventListener('input', () => {
            if (document.getElementById('tab-papers').classList.contains('active')) {
                applyPaperFilters();
            }
        });

        // Scroll listener for Floating Back To Top Button
        window.addEventListener('scroll', () => {
            const btn = document.getElementById('backToTopBtn');
            if (window.scrollY > 300) {
                btn.classList.remove('opacity-0', 'pointer-events-none');
            } else {
                btn.classList.add('opacity-0', 'pointer-events-none');
            }
        });

        /* ==========================================================================
           9. WINDOW ONLOAD EXECUTION
           ========================================================================== */
        window.onload = async function() {
            // Set saved theme preference
            const savedTheme = localStorage.getItem('spp_theme');
            if (savedTheme === 'dark') {
                document.documentElement.classList.add('dark');
            }

            // Populate Dropdowns & Translations
            populateSelectOptions();
            updateTranslations();

            // Load IndexedDB Data and Initial Render
            await initIndexedDB();
            await loadAllData();
        };
    </script>
